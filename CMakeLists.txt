if(AITSNAP_QT_MAJOR STREQUAL "6")
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The Qt 6 Windows 10/11 build requires x64.")
    endif()
    find_package(Qt6 6.12...<6.13 REQUIRED COMPONENTS Widgets)
elseif(AITSNAP_QT_MAJOR STREQUAL "5")
    find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets)
else()
    message(FATAL_ERROR "AITSNAP_QT_MAJOR must be 5 or 6; no automatic fallback is used.")
endif()

set(AITSNAP_EMBEDDED_MSI "" CACHE FILEPATH "Existing MSI to embed in the branded setup entry")
if(AITSNAP_EMBEDDED_MSI)
    if(NOT EXISTS "${AITSNAP_EMBEDDED_MSI}")
        message(FATAL_ERROR "AITSNAP_EMBEDDED_MSI must name an existing installer")
    endif()
    file(SHA256 "${AITSNAP_EMBEDDED_MSI}" AITSNAP_INSTALLER_SHA256)
    file(TO_CMAKE_PATH "${AITSNAP_EMBEDDED_MSI}" AITSNAP_MSI_RESOURCE_PATH)
    configure_file(installer/setup_config.h.in "${CMAKE_CURRENT_BINARY_DIR}/setup_config.h" @ONLY)
    configure_file(resources/setup-payload.rc.in "${CMAKE_CURRENT_BINARY_DIR}/setup-payload.rc" @ONLY)
    add_executable(aitsnap-setup WIN32 installer/setup_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/setup-payload.rc" "${CMAKE_CURRENT_BINARY_DIR}/aitsnap-version.rc")
    set_target_properties(aitsnap-setup PROPERTIES AUTOMOC OFF AUTORCC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_include_directories(aitsnap-setup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(aitsnap-setup PRIVATE UNICODE _UNICODE)
    target_link_libraries(aitsnap-setup PRIVATE shell32 bcrypt ole32 user32)
    if(MSVC)
        target_compile_options(aitsnap-setup PRIVATE /W4 /utf-8)
    endif()
endif()

foreach(target IN LISTS aitsnap_targets)
    target_compile_definitions(${target} PRIVATE AITSNAP_VERSION="${PROJECT_VERSION}")
    target_link_libraries(${target} PRIVATE Qt${AITSNAP_QT_MAJOR}::Widgets)
    if(AITSNAP_QT_MAJOR STREQUAL "5")
        # Header API floor only; this does not qualify compiler/CRT dependencies.
        target_compile_definitions(${target} PRIVATE WINVER=0x0601 _WIN32_WINNT=0x0601)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /Zc:__cplusplus)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
